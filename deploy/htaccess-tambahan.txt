# Aturan yang disisipkan pasang-di-hostingan.sh ke .htaccess hostingan.
# Jangan menyunting .htaccess di server tangan, sunting berkas ini lalu push.
#
# Kompresi SENGAJA tidak diatur di sini. Sudah diuji langsung di hostingan,
# brotli hidup bawaan dan berkas JSON 347 KB turun jadi 104 KB tanpa aturan
# apa pun. Menambah AddOutputFilterByType di sini cuma jadi berkas mati.
# Catatan, berkas kecil memang tidak dikompresi, itu wajar.

# =====================================================================
# PAKSA HTTPS, dan satukan ke satu alamat resmi.
#
# Dipasang 5 September 2026, sesudah AutoSSL menerbitkan Let's Encrypt untuk
# envirocircle.info. JANGAN dipasang sebelum sertifikatnya sah, kalau tidak
# semua pengunjung dilempar ke halaman peringatan merah.
#
# Kenapa perlu. Tanpa ini situsnya bisa dibuka lewat http maupun https, dan
# yang mengetik alamatnya tanpa https mendarat di http. Di situ browser
# menandainya "Not secure" DAN mematikan geolokasi, sebab fitur itu cuma
# jalan di sambungan aman. Jadi tombol Lokasi kelihatan rusak padahal
# kodenya benar.
#
# Dua aturan, urutannya penting.
# =====================================================================
<IfModule mod_rewrite.c>
  RewriteEngine On

  # 1. Alamat selain domain resmi dilempar ke domain resmi, sekalian https.
  #    Ini yang menangani alamat sementara penyedia dan alamat berbasis IP.
  #    Sengaja 302, BUKAN 301. Pengalihan permanen disimpan browser dan sulit
  #    ditarik kembali, dan alamat sementara itu masih berguna buat memeriksa
  #    sesuatu langsung ke server tanpa lewat DNS.
  RewriteCond %{HTTP_HOST} !^(www\.)?envirocircle\.info$ [NC]
  RewriteRule ^(.*)$ https://envirocircle.info/$1 [R=302,L]

  # 2. Sudah di domain resmi tapi masih http, naikkan ke https.
  #    Yang ini 301 dan memang seharusnya permanen. Sertifikatnya diperbarui
  #    AutoSSL sendiri, jadi tidak ada alasan turun lagi ke http.
  RewriteCond %{HTTPS} !=on
  RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]
</IfModule>

# Daftar isi folder MENYALA bawaan di hostingan ini. Tanpa baris ini, folder
# data yang tidak punya index membalas 200 dan memajang seluruh nama
# berkasnya ke siapa saja.
Options -Indexes

# Salinan kerja git ada di dalam folder yang disajikan, jadi tanpa ini siapa
# pun bisa mengunduh seluruh repo berikut riwayatnya lewat browser.
RedirectMatch 404 /\.git
RedirectMatch 404 /\.gitignore$

<IfModule mod_headers.c>
  # Data peta berganti tiap hari, jadi jangan disimpan lama lama di browser.
  # Sepuluh menit cukup untuk menahan muat ulang beruntun tanpa membuat orang
  # melihat data kemarin.
  <FilesMatch "\.(json|webp|png)$">
    Header set Cache-Control "public, max-age=600"
  </FilesMatch>

  # Kode situs. DITURUNKAN dari 3600 jadi 60 pada 5 September 2026, sebab
  # angka lamanya betul betul menggigit. Sesudah menarik perubahan ke
  # hostingan, browser tetap menyajikan app.js LAMA sampai satu jam, jadi
  # perubahan yang sudah mendarat kelihatan seperti tidak jadi. Sudah kena
  # sekali waktu mengganti sandi gerbang, dan bingungnya mahal.
  #
  # 60 detik itu pilihan untuk masa situsnya masih sering disunting. Kalau
  # nanti sudah jarang berubah, naikkan lagi ke 3600 supaya lebih hemat.
  # Ctrl+Shift+R tetap cara tercepat kalau tidak mau menunggu.
  <FilesMatch "\.(js|css)$">
    Header set Cache-Control "public, max-age=60"
  </FilesMatch>

  # HTML jangan pernah disimpan. Dia yang menunjuk ke berkas lain, jadi kalau
  # dia basi, seluruh situs ikut basi walaupun berkas lainnya sudah baru.
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
</IfModule>
